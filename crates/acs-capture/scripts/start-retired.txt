# Like start-scenario.txt, for a scenario disk with a retired player of ours
# (acs-capture scenario ... --player): picks that player instead of creating
# one (docs/capture.md section 8). Start VICE with --disk SCENARIO
# --snapshot boot/play-nodisk.vsf; saves ${OUT}/start.vsf at the first turn.
# Choosing the player saves the adventure: the disk changes.
reset 8              # the drive re-reads the new disk at a fixed point
run 5m
joy 1 fire
run 4m
joy 1 none
run 100m             # title screen
shot start-title
joy 1 fire
run 4m
joy 1 none
run 60m              # add player
shot start-add       # the retired player is listed, with the cursor on it
joy 1 fire
run 4m
joy 1 none
run 160m             # "press button of the joystick that will control..."
joy 1 fire
run 4m
joy 1 none
run 160m             # saving game
joy 1 fire
run 4m
joy 1 none
run 160m             # first turn
shot start-map
dump ${OUT}/start.vsf
